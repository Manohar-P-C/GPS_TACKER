ESP32 main.py code which gets connect to same hotspot both in controller and laptop


import network
import time
import json
import socket
from machine import UART, Pin

# ---------- WIFI SETTINGS ----------
WIFI_SSID = "Not"
WIFI_PASSWORD = "120120120"

# ---------- LAPTOP SERVER ----------
SERVER_IP = "10.209.79.38"
SERVER_PORT = 8000

# ---------- GPS UART ----------
gps = UART(
    2,
    baudrate=9600,
    tx=Pin(17),
    rx=Pin(16)
)


def connect_wifi():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)

    if wlan.isconnected():
        print("Wi-Fi connected:", wlan.ifconfig()[0])
        return wlan

    print("Connecting to hotspot...")
    wlan.connect(WIFI_SSID, WIFI_PASSWORD)

    for _ in range(30):
        if wlan.isconnected():
            print("Wi-Fi connected:", wlan.ifconfig()[0])
            return wlan
        time.sleep(1)

    print("Wi-Fi connection failed.")
    return wlan


def to_decimal(value, direction):
    if not value:
        return None

    degree_digits = 2 if direction in ("N", "S") else 3
    degrees = float(value[:degree_digits])
    minutes = float(value[degree_digits:])
    decimal = degrees + minutes / 60

    if direction in ("S", "W"):
        decimal = -decimal

    return decimal


def send_location(latitude, longitude):
    payload = json.dumps({
        "latitude": latitude,
        "longitude": longitude
    })

    request = (
        "POST /gps HTTP/1.1\r\n"
        "Host: {}:{}\r\n"
        "Content-Type: application/json\r\n"
        "Content-Length: {}\r\n"
        "Connection: close\r\n"
        "\r\n"
        "{}"
    ).format(
        SERVER_IP,
        SERVER_PORT,
        len(payload),
        payload
    )

    sock = None

    try:
        address = socket.getaddrinfo(
            SERVER_IP, SERVER_PORT
        )[0][-1]

        sock = socket.socket()
        sock.settimeout(4)
        sock.connect(address)
        sock.send(request.encode())

        response = sock.recv(256).decode("utf-8", "ignore")

        if "200 OK" in response:
            print("Location sent successfully!")
            print(latitude, longitude)
        else:
            print("Server response:", response[:120])

    except Exception as error:
        print("Send failed:", error)

    finally:
        if sock:
            sock.close()


wlan = connect_wifi()
last_sent = 0

print("GPS-to-laptop transmitter started.")

while True:
    # Reconnect if Wi-Fi disconnects
    if not wlan.isconnected():
        print("Wi-Fi disconnected. Reconnecting...")
        wlan = connect_wifi()

    if gps.any():
        line = gps.readline()

        if line:
            sentence = line.decode("ascii", "ignore").strip()
            fields = sentence.split(",")

            # Process GPS GGA sentences only
            if fields[0].endswith("GGA") and len(fields) > 8:
                try:
                    fix_quality = int(fields[6] or "0")

                    if fix_quality > 0 and fields[2] and fields[4]:
                        latitude = to_decimal(fields[2], fields[3])
                        longitude = to_decimal(fields[4], fields[5])

                        now = time.time()

                        # Send at most once every 2 seconds
                        if now - last_sent >= 2 and wlan.isconnected():
                            send_location(latitude, longitude)
                            last_sent = now

                except (ValueError, IndexError):
                    pass

    time.sleep_ms(50)
