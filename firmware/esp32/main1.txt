ESP32 main.py code which gets connect to same hotspot both in controller and laptop



import network
import time
import json
import urequests
from machine import UART, Pin
import config

# ---------- RENDER SERVER ----------
SERVER_URL = config.SERVER_URL.rstrip("/")
GPS_API_KEY = config.GPS_API_KEY

# ---------- GPS UART ----------
gps = UART(
    2,
    baudrate=9600,
    tx=Pin(17),
    rx=Pin(16)
)


def connect_wifi():
    wlan = network.WLAN(network.STA_IF)
    wlan.active(True)

    if wlan.isconnected():
        print("Wi-Fi connected:", wlan.ifconfig()[0])
        return wlan

    print("Connecting to Wi-Fi...")
    wlan.connect(config.WIFI_SSID, config.WIFI_PASSWORD)

    for _ in range(30):
        if wlan.isconnected():
            print("Wi-Fi connected:", wlan.ifconfig()[0])
            return wlan
        time.sleep(1)

    print("Wi-Fi connection failed.")
    return wlan


def to_decimal(value, direction):
    if not value:
        return None

    degree_digits = 2 if direction in ("N", "S") else 3
    degrees = float(value[:degree_digits])
    minutes = float(value[degree_digits:])
    decimal = degrees + minutes / 60

    if direction in ("S", "W"):
        decimal = -decimal

    return decimal


def send_location(latitude, longitude):
    url = SERVER_URL + "/gps"

    headers = {
        "Content-Type": "application/json",
        "X-API-Key": GPS_API_KEY
    }

    payload = {
        "latitude": latitude,
        "longitude": longitude
    }

    response = None

    try:
        response = urequests.post(
            url,
            data=json.dumps(payload),
            headers=headers
        )

        if response.status_code == 200:
            print("Location sent successfully!")
            print("Latitude:", latitude)
            print("Longitude:", longitude)
            return True

        print("Server HTTP status:", response.status_code)
        return False

    except Exception as error:
        print("Send failed:", error)
        return False

    finally:
        if response is not None:
            response.close()


wlan = connect_wifi()
last_sent = 0

print("GPS-to-Render transmitter started.")

while True:

    # Reconnect if Wi-Fi disconnects
    if not wlan.isconnected():
        print("Wi-Fi disconnected. Reconnecting...")
        wlan = connect_wifi()

    if wlan.isconnected() and gps.any():
        line = gps.readline()

        if line:
            sentence = line.decode("ascii", "ignore").strip()
            fields = sentence.split(",")

            # Process GPS GGA sentences only
            if fields[0].endswith("GGA") and len(fields) > 8:
                try:
                    fix_quality = int(fields[6] or "0")

                    if fix_quality > 0 and fields[2] and fields[4]:

                        latitude = to_decimal(fields[2], fields[3])
                        longitude = to_decimal(fields[4], fields[5])

                        now = time.time()

                        # Send at most once every 2 seconds
                        if now - last_sent >= 2:
                            success = send_location(
                                latitude,
                                longitude
                            )

                            if success:
                                last_sent = now

                except (ValueError, IndexError):
                    pass

    time.sleep_ms(50)
